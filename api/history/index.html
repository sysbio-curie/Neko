    History and network states¶

    History is implemented directly by Network; there is no separate NetworkHistory class. Mutating decorated methods save deep node/edge snapshots as NetworkState objects. Undo, redo, checkout, and subsequent mutations form a branching tree.

    For structured topology explanations, provenance, pruning gaps, and JSON export/recovery, see Interaction curation.

    net.connect_nodes()
states = net.list_states()
net.undo()
net.redo()
html = net.history_html()

    For routine inspection, avoid materializing complete history:

    status = net.history_status()
count = net.state_count
page = net.list_states(offset=0, limit=25, detail="summary")
details = net.get_state(page[0]["id"]) if page else None

    history_status() returns pruned_state_count and working_state_matches_current. It compares the working tables with the current snapshot, including nested provenance, without traversing older snapshots. It costs work proportional to the current network, not the retained history. state_count is the number of retained states.

    list_states(*, offset=0, limit=None, detail="full") slices IDs before copying metadata. The default result is unchanged. Summary entries retain id, parent_ids, children_ids and current; their metadata contains method, label, ancestry_gaps and transition. The summary transition includes explanation_status and counts, with nodes/edges counts for added, removed and changed. Evidence, parameters and inclusion explanations are available through get_state. Summary entries omit the repeated pruned_state_ids list; use history_status for the count and history_dict for exact IDs. All returned mutable values are independent copies. For legacy snapshots without a recorded table delta, that table's counts are None, rather than invented zeroes.

    Offsets and limits must be non-negative integers (booleans are rejected); limit=None selects the remaining states, and zero returns an empty list. Unknown detail modes and invalid pagination values raise ValueError. Pages use creation order; they are not frozen cursors across concurrent edits.

    MCP summaries can use the page's parent/child IDs directly, together with state_count and history_status. Calling history_graph() or history_dict() as well would reintroduce full-history work.

    def save_state(self, metadata: Optional[dict] = None) -> int:
    """Persist the current network snapshot and return its state id."""

    parent_id = self._current_state_id
    state = self._build_state(metadata, parent_id)
    state_id = state.state_id
    self._states[state_id] = state
    self._state_metadata[state_id] = state.metadata
    self._state_log.append(state_id)

    if parent_id is None:
        self._root_state_id = state_id
    else:
        parent_state = self._states[parent_id]
        parent_state.add_child(state_id)

    self._current_state_id = state_id
    self._state_counter += 1
    self._enforce_max_history()
    return state_id

    def list_states(self, *, offset=0, limit=None, detail="full") -> list:
    """Return a creation-ordered page; defaults preserve the full listing.

    Summary metadata includes labels, explanation status and change counts,
    without copying evidence, parameters or inclusion explanations.
    """
    if type(offset) is not int or offset < 0:
        raise ValueError("offset must be a non-negative integer.")
    if limit is not None and (type(limit) is not int or limit < 0):
        raise ValueError("limit must be None or a non-negative integer.")
    if detail not in ("full", "summary"):
        raise ValueError("detail must be 'full' or 'summary'.")
    stop = None if limit is None else offset + limit
    result = []
    for state_id in self._state_log[offset:stop]:
        metadata = self._state_metadata.get(state_id, {})
        if detail == "summary":
            transition = metadata.get("transition", {})
            metadata = {
                "method": metadata.get("method"),
                "label": metadata.get("label"),
                "ancestry_gaps": list(metadata.get("ancestry_gaps", [])),
                "transition": {
                    "explanation_status": transition.get("explanation_status", "unavailable"),
                    "counts": {
                        table: {kind: len(transition.get(table, {}).get(kind, []))
                                for kind in ("added", "removed", "changed")}
                        if table in transition else None
                        for table in ("nodes", "edges")
                    },
                },
            }
        entry = {
            "id": state_id, "metadata": copy.deepcopy(metadata),
            "parent_ids": list(self._states[state_id].parent_ids),
            "children_ids": list(self._states[state_id].children_ids),
            "current": state_id == self.current_state_id,
        }
        if detail == "full":
            entry["pruned_state_ids"] = list(self._pruned_state_ids)
        result.append(entry)
    return result

    def undo(self) -> None:
    """Move to the parent state if available."""

    if self._current_state_id is None:
        return
    parents = self._get_state(self._current_state_id).parent_ids
    if not parents:
        return
    self.checkout(parents[-1])

    def redo(self, state_id: Optional[int] = None) -> None:
    """Move to a child state. If more than one child exists, a specific state id is required."""

    if self._current_state_id is None:
        return
    children = self._get_state(self._current_state_id).children_ids
    if not children:
        return
    target = state_id
    if target is None:
        if len(children) != 1:
            raise ValueError("Multiple branches available; specify a target state id.")
        target = children[0]
    elif target not in children:
        raise ValueError(f"State {target} is not a child of {self._current_state_id}.")
    self.checkout(target)

    def compare_states(self, state_a: int, state_b: int) -> dict:
    """Compare two states identified by their ids."""

    resolved_a = self._resolve_state_id(state_a)
    resolved_b = self._resolve_state_id(state_b)
    state1 = self._get_state(resolved_a)
    state2 = self._get_state(resolved_b)
    nodes1 = self._state_node_labels(state1)
    nodes2 = self._state_node_labels(state2)
    edges1 = self._state_edge_signatures(state1)
    edges2 = self._state_edge_signatures(state2)
    diff = {
        "added_nodes": list(nodes2 - nodes1),
        "removed_nodes": list(nodes1 - nodes2),
        "added_edges": list(edges2 - edges1),
        "removed_edges": list(edges1 - edges2),
        "node_changes": _history.delta(state1.nodes, state2.nodes, ["Uniprot", "Genesymbol"]),
        "edge_changes": _history.delta(state1.edges, state2.edges, ["source", "target"]),
    }
    return diff

    def describe_history(self) -> None:
    """Pretty-print the branching history tree."""

    if not self._states:
        print("<no states recorded>")
        return

    def _label(state: NetworkState) -> str:
        meta = state.metadata or {}
        label = meta.get("label") or meta.get("description")
        if label:
            return str(label)
        meta = {key: value for key, value in meta.items() if key not in _history.INTERNAL_METADATA_KEYS}
        if meta:
            return str(meta)
        return ""

    def _walk(state_id: int, depth: int) -> None:
        state = self._get_state(state_id)
        indent = "  " * depth
        label = _label(state)
        suffix = f" - {label}" if label else ""
        print(f"{indent}State {state_id}{suffix}")
        for child_id in state.children_ids:
            _walk(child_id, depth + 1)

    _walk(self._root_state_id, 0)

    def build_history_graph(network: "Network") -> nx.DiGraph:
    """Return a directed graph of the network state history."""

    graph = nx.DiGraph()
    graph.graph.update(database_name=network.database_name,
                       current_state_id=network.current_state_id,
                       pruned_state_ids=list(network._pruned_state_ids))
    for state_id in network.list_states():
        # list_states returns dictionaries with id/metadata
        graph.add_node(state_id["id"], **state_id.get("metadata", {}))
    for state_id, state in network._states.items():  # pylint: disable=protected-access
        for child in state.children_ids:
            graph.add_edge(state_id, child)
    return graph

    def history_digraph(network: "Network", include_metadata: bool = True) -> Digraph:
    """Render history as a Graphviz Digraph."""

    digraph = Digraph()

    for state_id, state in network._states.items():  # pylint: disable=protected-access
        metadata = state.metadata or {}
        label_lines = [f"State {state_id}"]
        if metadata.get("label"):
            label_lines.append(str(metadata["label"]))
        if state_id == network.current_state_id:
            label_lines.append("current")
        if include_metadata and metadata.get("method"):
            args = metadata.get("args", [])
            kwargs = metadata.get("kwargs", {})
            params = ", ".join(args)
            if kwargs:
                kwargs_repr = ", ".join(f"{k}={v}" for k, v in kwargs.items())
                params = ", ".join(filter(None, [params, kwargs_repr]))
            label_lines.append(f"{metadata['method']}({params[:160]})")
        elif include_metadata:
            extras = {
                key: value for key, value in metadata.items()
                if key not in {"label", "method", "args", "kwargs"}
                and (key == "database_name" or key not in INTERNAL_METADATA_KEYS)
            }
            for key, value in extras.items():
                label_lines.append(f"{key}: {value}")
        if metadata.get('ancestry_gaps'):
            label_lines.append('ancestry gap (pruned states)')
        digraph.node(str(state_id), "\n".join(label_lines),
                     URL=f"#neko-state-{state_id}" if include_metadata else "",
                     tooltip=f"State {state_id}: structured explanation")
        for child in state.children_ids:
            digraph.edge(str(state_id), str(child))
    return digraph

    def history_html(network: "Network", include_metadata: bool = True, div_class: str = "neko-history-graph") -> str:
    """Return an HTML snippet embedding the history graph as SVG."""

    svg = history_digraph(network, include_metadata=include_metadata).pipe(format="svg").decode("utf-8")
    details = ''
    if include_metadata:
        details = ''.join(
            f'<details id="neko-state-{sid}"><summary>State {sid} explanation</summary>'
            f'<pre>{html.escape(json.dumps(json_value(state.metadata), indent=2))}</pre></details>'
            for sid, state in network._states.items()
        )
    return f'<div class="{html.escape(div_class, quote=True)}">{svg}{details}</div>'
